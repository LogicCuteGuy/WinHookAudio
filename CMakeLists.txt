cmake_minimum_required(VERSION 3.24)
project(WinHookAudio VERSION 0.1.0 LANGUAGES CXX)

if(NOT WIN32 OR NOT MSVC OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
    message(FATAL_ERROR "WinHookAudio currently requires Windows, MSVC, and x64.")
endif()

add_library(wha_common INTERFACE)
target_include_directories(wha_common INTERFACE ${CMAKE_CURRENT_SOURCE_DIR}/common)
target_compile_features(wha_common INTERFACE cxx_std_20)

add_executable(winhookaudio-probe
    src/probe/main.cpp
    src/probe/mmdevice.cpp
    src/probe/ks.cpp
    src/probe/asio_registry.cpp)
target_compile_features(winhookaudio-probe PRIVATE cxx_std_20)
set_property(TARGET winhookaudio-probe PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
target_compile_definitions(winhookaudio-probe PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN)
target_compile_options(winhookaudio-probe PRIVATE /W4 /WX /permissive- /utf-8 /EHsc)
target_link_libraries(winhookaudio-probe PRIVATE ole32 propsys setupapi ksuser advapi32 uuid wha_common)
add_executable(winhookaudio-abi-check src/abi-check/main.cpp)
target_compile_features(winhookaudio-abi-check PRIVATE cxx_std_20)
set_property(TARGET winhookaudio-abi-check PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
target_compile_definitions(winhookaudio-abi-check PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN)
target_compile_options(winhookaudio-abi-check PRIVATE /W4 /WX /permissive- /utf-8 /EHsc)
target_link_libraries(winhookaudio-abi-check PRIVATE wha_common)

# 08 Master minimal — offline DLLs (no SDK required, stub IASIO)
add_library(winhookaudio-master SHARED asio-master/WinHookMasterASIO.cpp asio-master/MasterHolder.cpp asio-master/KsAudio.cpp asio-master/DllMain.cpp)
target_compile_features(winhookaudio-master PRIVATE cxx_std_20)
set_property(TARGET winhookaudio-master PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
target_compile_definitions(winhookaudio-master PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN)
target_compile_options(winhookaudio-master PRIVATE /W4 /WX /permissive- /utf-8 /EHsc)
target_include_directories(winhookaudio-master PRIVATE common)
target_link_libraries(winhookaudio-master PRIVATE ole32 uuid wha_common)
set_target_properties(winhookaudio-master PROPERTIES OUTPUT_NAME "WinHookAudioMasterASIO64")

add_library(winhookaudio-bridge SHARED asio-bridge/WinHookBridgeASIO.cpp asio-bridge/DllMain.cpp)
target_compile_features(winhookaudio-bridge PRIVATE cxx_std_20)
set_property(TARGET winhookaudio-bridge PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
target_compile_definitions(winhookaudio-bridge PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN)
target_compile_options(winhookaudio-bridge PRIVATE /W4 /WX /permissive- /utf-8 /EHsc)

set_target_properties(winhookaudio-bridge PROPERTIES OUTPUT_NAME "WinHookAudioBridgeASIO64")

# SDK host sample offline test (CoCreateInstance -> init -> getChannels -> getChannelInfo -> createBuffers -> start -> bufferSwitch)
add_executable(winhookaudio-host-sample tests/host-sample/main.cpp asio-master/WinHookMasterASIO.cpp asio-master/MasterHolder.cpp asio-master/KsAudio.cpp asio-bridge/WinHookBridgeASIO.cpp)

install(TARGETS winhookaudio-probe RUNTIME DESTINATION bin)
install(FILES docs/probe-vm.md DESTINATION . RENAME README.md)
install(FILES scripts/Run-Probe.ps1 DESTINATION .)
