# 15 WHAA codecs (ADR 0009): real libvorbis/libogg/r8brain when vendored under third_party/, stub otherwise.
add_library(wha_codecs INTERFACE)
set(WHA_OGG_DIR ${CMAKE_CURRENT_SOURCE_DIR}/third_party/libogg CACHE PATH "Vendored libogg 1.3.5")
set(WHA_VORBIS_DIR ${CMAKE_CURRENT_SOURCE_DIR}/third_party/libvorbis CACHE PATH "Vendored libvorbis 1.3.7")
set(WHA_R8B_DIR ${CMAKE_CURRENT_SOURCE_DIR}/third_party/r8brain CACHE PATH "Vendored r8brain-free 7.5")
if(EXISTS ${WHA_OGG_DIR}/include/ogg/ogg.h AND EXISTS ${WHA_VORBIS_DIR}/include/vorbis/codec.h)
    enable_language(C)
    add_library(wha_ogg STATIC ${WHA_OGG_DIR}/src/bitwise.c ${WHA_OGG_DIR}/src/framing.c)
    target_include_directories(wha_ogg SYSTEM PUBLIC ${WHA_OGG_DIR}/include)
    set(WHA_VORBIS_SOURCES mdct.c smallft.c block.c envelope.c window.c lsp.c lpc.c analysis.c synthesis.c
        psy.c info.c floor1.c floor0.c res0.c mapping0.c registry.c codebook.c sharedbook.c lookup.c bitrate.c vorbisenc.c)
    list(TRANSFORM WHA_VORBIS_SOURCES PREPEND ${WHA_VORBIS_DIR}/lib/)
    add_library(wha_vorbis STATIC ${WHA_VORBIS_SOURCES})
    target_include_directories(wha_vorbis SYSTEM PUBLIC ${WHA_VORBIS_DIR}/include PRIVATE ${WHA_VORBIS_DIR}/lib)
    target_link_libraries(wha_vorbis PUBLIC wha_ogg)
    foreach(t wha_ogg wha_vorbis)
        set_property(TARGET ${t} PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
        target_compile_definitions(${t} PRIVATE _CRT_SECURE_NO_WARNINGS _CRT_NONSTDC_NO_DEPRECATE)
        target_compile_options(${t} PRIVATE /W1)
    endforeach()
    target_link_libraries(wha_codecs INTERFACE wha_vorbis)
    target_compile_definitions(wha_codecs INTERFACE WHA_HAVE_VORBIS=1)
    message(STATUS "WHAA Vorbis: libvorbis + libogg vendored")
else()
    message(STATUS "WHAA Vorbis: stub (third_party/libvorbis + third_party/libogg not vendored)")
endif()
if(EXISTS ${WHA_R8B_DIR}/CDSPResampler.h)
    target_include_directories(wha_codecs SYSTEM INTERFACE ${WHA_R8B_DIR})
    target_compile_definitions(wha_codecs INTERFACE WHA_HAVE_R8BRAIN=1)
    message(STATUS "WHAA SRC: r8brain vendored")
else()
    message(STATUS "WHAA SRC: stub (third_party/r8brain not vendored)")
endif()
